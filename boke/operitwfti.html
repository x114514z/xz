detect_pkg_manager() {
    if   command -v apt-get &>/dev/null; then echo "apt"
    elif command -v dnf     &>/dev/null; then echo "dnf"
    elif command -v yum     &>/dev/null; then echo "yum"
    elif command -v zypper  &>/dev/null; then echo "zypper"
    elif command -v pacman  &>/dev/null; then echo "pacman"
    elif command -v pkg     &>/dev/null; then echo "pkg"
    else echo "unknown"
    fi
}

install_pkgs() {
    local pm="$1"; shift
    case "$pm" in
        apt)    sudo apt-get update -qq && sudo apt-get install -y "$@" ;;
        dnf)    sudo dnf install -y "$@" ;;
        yum)    sudo yum install -y "$@" ;;
        zypper) sudo zypper --non-interactive install "$@" ;;
        pacman) sudo pacman -Sy --noconfirm "$@" ;;
        pkg)    sudo pkg install -y "$@" ;;
        *)      error "不支持的包管理器: $pm" ;;
    esac
}

            install_pkgs apt unzip wget curl default-jdk-headless ;;
    dnf|yum) install_pkgs "$PM" unzip wget curl java-17-openjdk-headless 2>/dev/null || \
             install_pkgs "$PM" unzip wget curl java-latest-openjdk-headless ;;
    zypper) install_pkgs zypper unzip wget curl java-17-openjdk-headless ;;
    pacman) install_pkgs pacman unzip wget curl jdk-openjdk ;;
    pkg)    install_pkgs pkg unzip wget curl openjdk17 ;;

    fi

    rm -rf "$TMP_EXTRACT"
    unzip -q "$TMP_ZIP" -d "$TMP_EXTRACT"

    # 标准结构：cmdline-tools/latest/bin/sdkmanager
    rm -rf "$CMDLINE_DIR"
    mv "$TMP_EXTRACT/cmdline-tools" "$CMDLINE_DIR"

# ============================

    echo 请先安装 Android SDK Platform-Tools，并将 adb 所在目录加入系统 PATH。

    set /p open_url=
    if /i "!open_url!"=="Y" start https://developer.android.com/studio/releases/platform-tools
    pause
    exit /b 1
)
echo [OK] 已检测到 ADB。
adb version
echo.

:: ============================================================
:: 2. 无线调试连接引导
:: ============================================================
echo ============================================================
echo                    无线调试连接
echo ============================================================
echo 请选择连接方式：
echo   [1] 手机已通过 USB 连接（推荐，脚本自动启用无线调试）
echo   [2] 使用“配对码”无线连接（Android 11+，无需 USB）
echo   [3] 已手动无线连接，直接跳过

    adb devices
    echo.

    adb tcpip 5555
    echo.
    set /p phone_ip=请输入手机的 Wi-Fi IP 地址（例如 192.168.1.100）：
    echo 正在连接 !phone_ip!:5555 ...
    adb connect !phone_ip!:5555
    echo.
    echo [提示] 现在可以拔掉 USB 线，后续操作将通过无线进行。

    set /p pair_port=请输入配对端口（如 37123）：
    set /p pair_code=请输入 6 位配对码：
    echo 正在配对...
    adb pair !pair_ip!:!pair_port! !pair_code!
    echo.
    echo 配对成功后，请再输入无线调试端口（无线调试主界面显示的“IP 地址和端口”中的端口号）：

        if command -v apt >/dev/null 2>&1; then sudo apt install -y android-tools-adb
        elif command -v dnf >/dev/null 2>&1; then sudo dnf install -y android-tools
        elif command -v pacman >/dev/null 2>&1; then sudo pacman -S --noconfirm android-tools
        elif command -v zypper >/dev/null 2>&1; then sudo zypper install -y android-tools
        else err "无法识别包管理器，请手动安装。"; exit 1; fi
